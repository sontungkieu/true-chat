\begin{frame}{Context policy và action space}
  \scriptsize
  \renewcommand{\arraystretch}{1.12}
  \begin{tabular}{>{\raggedright\arraybackslash}p{0.22\textwidth}>{\raggedright\arraybackslash}p{0.70\textwidth}}
    \toprule
    Term & Định nghĩa trong project \\
    \midrule
    \code{legacy/full} & Giữ rank order và nối block context; \code{full} nghĩa là không đặt action budget riêng, chỉ chịu ceiling prompt/top-k đang có. \\
    \code{char-budget} & Lấp một global character budget theo rank order; trim block cuối nếu vượt budget. \\
    \code{per-doc-budget} & Cắt từng document trước rồi mới lấp global budget để tránh một document dài chiếm toàn bộ context. \\
    \code{score-density} & Ưu tiên \code{retrieval\_score / estimated\_tokens}; fallback \code{1 / estimated\_tokens} nếu thiếu score. \\
    \code{sentence-trim} & Giữ rank order nhưng trim tại ranh giới câu đơn giản khi có thể. \\
    \code{evidence-aware} & Tên CLI tương thích cũ; implementation hiện tại là \code{lexical-query-aware}: tách span, chấm bằng query overlap + retrieval score + title overlap. \\
    \code{adaptive-heuristic} & Selector theo luật chọn fixed policy + budget dựa trên score gap, entropy và document length; đây là baseline trước bandit, chưa phải RL. \\
    \bottomrule
  \end{tabular}

  \vspace{0.28cm}
  \footnotesize
  \minihead{Profiles:}
  \textbf{conservative} ưu tiên budget lớn khi retrieval không chắc;
  \textbf{balanced} tách ngân sách theo score gap/entropy;
  \textbf{aggressive} stress-test budget nhỏ hơn.
\end{frame}

\begin{frame}{Reward: tính cụ thể từng thành phần}
  \footnotesize
  \[
    R = w_q Q_{answer} + w_s S_{evident}
    - w_t C_{tok} - w_l C_{lat}
    - w_k C_{kv} - w_e P_{error} - w_u P_{unsupported}
  \]

  \vspace{-0.1cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.50\textwidth}
      \minihead{Quality/support từ RLAIF}\\[-0.1cm]
      \begin{tightitemize}
        \item \(Q_{answer}\): \smallcode{overall\_quality} của answer judge; nếu thiếu thì dùng proxy reward và ghi rõ.
        \item \(S_{evident}\): trung bình các tín hiệu \smallcode{evidence\_support} và \smallcode{context\_quality\_score}.
        \item \(P_{unsupported}\): \smallcode{unsupported\_claim\_penalty} trong \([0,1]\).
        \item \(P_{error}\): 1 nếu generation/API/parsing lỗi, ngược lại 0.
      \end{tightitemize}
    \end{column}
    \begin{column}{0.47\textwidth}
      \minihead{Cost chuẩn hóa trong run}\\[-0.1cm]
      \begin{tightitemize}
        \item \(C_{tok}\): normalize \smallcode{prompt\_tokens} hoặc \smallcode{kept\_est\_tokens}.
        \item \(C_{lat}\): normalize \smallcode{answer\_latency\_s}.
        \item \(C_{kv}\): normalize \smallcode{estimated\_kv\_cache\_mb\_after}.
        \item \(norm(x)=\frac{x-\min(x)}{\max(x)-\min(x)+\epsilon}\), theo dataset/model/run.
      \end{tightitemize}
    \end{column}
  \end{columns}
  \takeaway{\scriptsize Default weights: \(w_q=1.0,w_s=0.2,w_t=0.2,w_l=0.2,w_k=0.1,w_e=1.0,w_u=1.0\). Tất cả nên là tham số CLI.}
\end{frame}

\begin{frame}{Offline contextual bandit: thuật toán}
  \small
  \begin{center}
    \normalsize
    Với mỗi query, policy chọn một action. Reward đến từ logged outcome; không tương tác online với người dùng.
  \end{center}

  \vspace{0.28cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.31\textwidth}
      \minihead{Input row}\\
      \color{Muted}query id, state features, action id, outcome metrics, optional RLAIF labels.
    \end{column}
    \begin{column}{0.31\textwidth}
      \minihead{Policies}\\
      \color{Muted}FixedAction, Cheapest, BestAverage, OracleLogged, epsilon-greedy replay, UCB, linear reward model.
    \end{column}
    \begin{column}{0.31\textwidth}
      \minihead{Evaluation}\\
      \color{Muted}split train/test theo \code{query\_id}; report reward, quality, token, latency, KV, error rate và action distribution.
    \end{column}
  \end{columns}

  \vspace{0.34cm}
  \footnotesize
  \begin{tabular}{>{\bfseries}p{0.18\textwidth}p{0.72\textwidth}}
    1. Observe & tính state \(s\) trước generation từ query và retrieval candidates. \\
    2. Choose & chọn \(a=(retriever, fusion, context\ policy, budget, profile)\). \\
    3. Score & build reward từ RLAIF/cost/error. \\
    4. Learn & update policy offline; nếu action chưa quan sát đủ thì report limitation. \\
  \end{tabular}
\end{frame}

\begin{frame}{Triển khai RLAIF và bandit trong repo}
  \small
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \minihead{RLAIF path}\\
      \tagline{theo Phase 1D plan}
      \begin{tightitemize}
        \item \smallcode{rlaif\_schema.py}: validate context/answer judge JSON.
        \item \smallcode{label\_contexts\_with\_rlaif.py}: chọn evidence tối thiểu, sufficiency, minimality.
        \item \smallcode{label\_answers\_with\_rlaif.py}: correctness, support, hallucination/refusal/citation.
        \item thiếu credential thì bỏ qua có log rõ ràng; không in secret.
      \end{tightitemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \minihead{Bandit path}\\
      \tagline{logged data trước, policy sau}
      \begin{tightitemize}
        \item \smallcode{build\_budgetrag\_logged\_dataset.py}: gom state/action/outcome.
        \item \smallcode{build\_budgetrag\_rewards.py}: normalize cost và merge RLAIF labels.
        \item \smallcode{retrieval\_context\_actions.py}: định nghĩa action set.
        \item \smallcode{train/evaluate\_budget\_bandit.py}: baselines, replay, UCB, linear reward model.
      \end{tightitemize}
    \end{column}
  \end{columns}
  \takeaway{\small Đây là contextual bandit/RL-lite. Không gọi là full RL nếu chưa có trajectory nhiều bước hoặc online policy improvement.}
\end{frame}

\begin{frame}{TurboQuant 16k/128k: tính payload chính xác}
  \scriptsize
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.37\textwidth}
      \minihead{Công thức payload}\\
      \[
        TQ_{3.5b}=KV_{bf16}\times\frac{3+4}{16+16}
        =KV_{bf16}\times\frac{7}{32}
      \]
      \vspace{0.12cm}
      \footnotesize\color{Muted}
      Giả định variant 3-bit key + 4-bit value. Đây chỉ là payload KV, chưa cộng metadata, packing alignment, kernel workspace hay allocator overhead.
      \vspace{0.28cm}

      \minihead{Ví dụ kiểm tra}\\
      \footnotesize\color{Muted}
      Qwen2.5-14B 16k bf16 KV = 3072 MiB.\\
      \(3072\times7/32=672\) MiB.
    \end{column}
    \begin{column}{0.59\textwidth}
      \minihead{Qwen2.5 KV payload nếu TurboQuant 3.5-bit}
      \vspace{0.12cm}
      \renewcommand{\arraystretch}{1.14}
      \setlength{\tabcolsep}{4pt}
      \begin{tabular}{l r r}
        \toprule
        Model & 16k TQ payload & 128k TQ payload \\
        \midrule
        Qwen2.5-0.5B & 42 MiB & 336 MiB \\
        Qwen2.5-1.5B & 98 MiB & 784 MiB \\
        Qwen2.5-3B & 126 MiB & 1008 MiB \\
        Qwen2.5-7B & 196 MiB & 1568 MiB \\
        Qwen2.5-14B & 672 MiB & 5376 MiB \\
        \bottomrule
      \end{tabular}\par
      \vspace{0.22cm}
      \footnotesize\color{Muted}
      128k Qwen2.5-14B: \(24576\times7/32=5376\) MiB. Không dùng headline 6x hoặc runtime savings khi chưa đo implementation cụ thể.
    \end{column}
  \end{columns}
\end{frame}
